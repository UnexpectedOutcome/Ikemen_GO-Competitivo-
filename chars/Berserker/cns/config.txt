[State -3, HAモーションに打撃無敵付加]

;************************************************
; ※ 以下にHA処理を行いたいステートの条件を羅列する
;     例ではステート全体、ステートの途中から最後まで、ステートの中央部の各箇所へ
;     HA処理を適用する場合の記述方法を示しています。
;************************************************
trigger1 = StateNo = 1100
trigger1 = AnimElemTime(1) >= 0
trigger1 = AnimElemTime(5) < 0  ; ステート1100の1コマ～4コマの間をHA化
;************************************************

ignorehitpause = 1

; HAの最終フレームに打撃がHITした場合は、どうしてもヘルパー、本体の順番にダメージを2重に食らってしまう。
; その為、最終フレーム 且つ 打撃を受けていない(Var(41) = 0)の場合は
; HAの最終フレームの後に1Fだけ打撃無敵を付加する。
; こうすることでHAの最終フレームに打撃がHITした場合でも、2重に食らうことが無くなる。
; 但し、HA中に攻撃を何も食らわなかった時でも、最終フレーム後の1F打撃無敵は出てしまうので要注意。
[State -3, HAモーション最終フレームから1F打撃無敵付加]
type = NotHitBy
triggerall = Helper(8800), Var(41) = 0
;************************************************
; ※ 以下にHA処理の終了フレーム条件を羅列します。
;     上記「HAモーションに打撃無敵付加」の処理で指定した範囲の最終フレームをここで指定します。
;     例では上記3パターンに対応する終了フレームの記述方法を示しています。
;************************************************
trigger1 = StateNo = 1100
trigger1 = AnimElemTime(5) = 0 ; ステート1100の1コマ～4コマの間をHA化
;************************************************
value = , AA, AP
time = 1
ignorehitpause = 1
